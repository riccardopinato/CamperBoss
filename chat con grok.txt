4. Login Screen con Google + Facebook (lib/features/auth/presentation/screens/login_screen.dart)dart

import 'package:easy_localization/easy_localization.dart';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:flutter/material.dart';
import 'package:google_sign_in/google_sign_in.dart';
import 'package:flutter_facebook_auth/flutter_facebook_auth.dart';

class LoginScreen extends StatelessWidget {
  const LoginScreen({super.key});

  Future<void> _signInWithGoogle(BuildContext context) async {
    try {
      final GoogleSignInAccount? googleUser = await GoogleSignIn().signIn();
      if (googleUser == null) return;

      final GoogleSignInAuthentication googleAuth = await googleUser.authentication;
      final credential = GoogleAuthProvider.credential(
        accessToken: googleAuth.accessToken,
        idToken: googleAuth.idToken,
      );

      await FirebaseAuth.instance.signInWithCredential(credential);
      // Navigator.pushReplacement to Home
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(e.toString())));
    }
  }

  Future<void> _signInWithFacebook(BuildContext context) async {
    try {
      final LoginResult loginResult = await FacebookAuth.instance.login();
      if (loginResult.status == LoginStatus.success) {
        final OAuthCredential credential = FacebookAuthProvider.credential(loginResult.accessToken!.tokenString);
        await FirebaseAuth.instance.signInWithCredential(credential);
      }
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(e.toString())));
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Padding(
        padding: const EdgeInsets.all(24.0),
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            const Icon(Icons.directions_car, size: 100, color: Colors.green),
            const SizedBox(height: 20),
            Text("app_name".tr(), style: const TextStyle(fontSize: 32, fontWeight: FontWeight.bold)),
            const SizedBox(height: 8),
            Text("welcome".tr(), textAlign: TextAlign.center, style: const TextStyle(fontSize: 18)),
            const SizedBox(height: 50),

            ElevatedButton.icon(
              onPressed: () => _signInWithGoogle(context),
              icon: const Icon(Icons.g_mobiledata),
              label: Text("continue_with_google".tr()),
              style: ElevatedButton.styleFrom(minimumSize: const Size(double.infinity, 56)),
            ),
            const SizedBox(height: 16),

            ElevatedButton.icon(
              onPressed: () => _signInWithFacebook(context),
              icon: const Icon(Icons.facebook),
              label: Text("continue_with_facebook".tr()),
              style: ElevatedButton.styleFrom(minimumSize: const Size(double.infinity, 56), backgroundColor: Colors.blue),
            ),

            const SizedBox(height: 30),
            const Text("o", style: TextStyle(fontSize: 18)),
            const SizedBox(height: 30),

            // Email/Password (da espandere dopo)
            TextButton(onPressed: () {}, child: Text("login".tr())),
            TextButton(onPressed: () {}, child: Text("register".tr())),
          ],
        ),

import 'dart:convert';
import 'package:http/http.dart' as http;
import 'package:geolocator/geolocator.dart';

class WeatherService {
  static const String baseUrl = 'https://api.open-meteo.com/v1/forecast';

  Future<Map<String, dynamic>> getWeather(double lat, double lon) async {
    final url = Uri.parse(
      '$baseUrl?latitude=$lat&longitude=$lon'
      '&current=temperature_2m,relative_humidity_2m,apparent_temperature,weather_code,wind_speed_10m'
      '&daily=weather_code,temperature_2m_max,temperature_2m_min,precipitation_probability_max'
      '&timezone=auto',
    );

    final response = await http.get(url);

    if (response.statusCode == 200) {
      return json.decode(response.body);
    } else {
      throw Exception('Errore nel caricamento meteo');
    }
  }

  // Traduzione codice meteo (WMO)
  String getWeatherDescription(int code) {
    switch (code) {
      case 0: return "Cielo sereno";
      case 1: case 2: case 3: return "Parzialmente nuvoloso";
      case 45: case 48: return "Nebbia";
      case 51: case 53: case 55: return "Pioggerella";
      case 61: case 63: case 65: return "Pioggia";
      case 71: case 73: case 75: return "Nevicata";
      case 80: case 81: case 82: return "Rovesci";
      default: return "Variabile";
    }
  }
}

class _WeatherCardState extends State<WeatherCard> {
  final WeatherService _weatherService = WeatherService();
  Map<String, dynamic>? weatherData;
  bool isLoading = true;

  @override
  void initState() {
    super.initState();
    _loadWeather();
  }

  Future<void> _loadWeather() async {
    try {
      Position position = await Geolocator.getCurrentPosition();
      final data = await _weatherService.getWeather(position.latitude, position.longitude);
      setState(() {
        weatherData = data;
        isLoading = false;
      });
    } catch (e) {
      setState(() => isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    if (isLoading) {
      return const Card(child: Center(child: CircularProgressIndicator()));
    }

    if (weatherData == null) {
      return const Card(child: Text("Errore caricamento meteo"));
    }

    final current = weatherData!['current'];
    final daily = weatherData!['daily'];

    return Card(
      elevation: 4,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(20)),
      child: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          children: [

              mainAxisAlignment: MainAxisAlignment.spaceBetween,
              children: [
                Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text("${current['temperature_2m']}°C",

import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';

class FirestoreService {
  final FirebaseFirestore _db = FirebaseFirestore.instance;
  final String? userId = FirebaseAuth.instance.currentUser?.uid;

  // Salva un viaggio
  Future<void> saveTrip(Map<String, dynamic> tripData) async {
    if (userId == null) return;
    await _db.collection('users').doc(userId).collection('trips').add({
      ...tripData,
      'createdAt': FieldValue.serverTimestamp(),
    });
  }

  // Salva una nota di journal
  Future<void> saveJournalEntry(Map<String, dynamic> entry) async {
    if (userId == null) return;
    await _db.collection('users').doc(userId).collection('journal').add({
      ...entry,
      'createdAt': FieldValue.serverTimestamp(),
    });
  }

  Stream<QuerySnapshot> getUserTrips() {
    return _db.collection('users').doc(userId).collection('trips').orderBy('createdAt', descending: true).snapshots();
  }
}

import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;
import 'package:latlong2/latlong.dart';

class IOverlanderService {
  static const String baseUrl = "http://app.ioverlander.com/countries";

  // Esempi: italy, germany, france, spain, united_states, ecc.
  Future<List<IOverlanderPlace>> getPlacesByCountry(String countryCode) async {
    try {
      final response = await http.get(Uri.parse("$baseUrl/places_by_country/$countryCode.json"));

      if (response.statusCode == 200) {
        final List<dynamic> data = json.decode(response.body);
        return data.map((json) => IOverlanderPlace.fromJson(json)).toList();
      } else {
        throw Exception("Errore ${response.statusCode}");
      }
    } catch (e) {
      debugPrint("Errore iOverlander: $e");
      return [];
    }
  }

  // Scarica e filtra solo i posti rilevanti per camper
  Future<List<IOverlanderPlace>> getCamperPlaces(String countryCode) async {
    final allPlaces = await getPlacesByCountry(countryCode);
    return allPlaces.where((place) =>
      place.placeType.toLowerCase().contains("camp") ||
      place.placeType.toLowerCase().contains("wild") ||
      place.placeType.toLowerCase().contains("dump") ||
      place.placeType.toLowerCase().contains("parking")
    ).toList();
  }
}

class IOverlanderPlace {
  final double lat;
  final double lng;
  final String name;
  final String placeType;
  final String? description;
  final String? creator;
  final int? rating;

  IOverlanderPlace({
    required this.lat,
    required this.lng,
    required this.name,
    required this.placeType,
    this.description,
    this.creator,
    this.rating,
  });

  factory IOverlanderPlace.fromJson(Map<String, dynamic> json) {
    return IOverlanderPlace(
      lat: double.parse(json['latitude'].toString()),
      lng: double.parse(json['longitude'].toString()),
      name: json['name'] ?? 'Senza nome',
      placeType: json['place_type'] ?? json['type'] ?? 'Unknown',
      description: json['description'],
      creator: json['creator'],
      rating: json['rating'] != null ? int.tryParse(json['rating'].toString()) : null,
    );
  }

  LatLng get latLng => LatLng(lat, lng);
}

import 'package:latlong2/latlong.dart';
import '../database/app_database.dart';
import '../../core/services/ioverlander_service.dart';

class IOverlanderRepository {
  final AppDatabase db;
  final IOverlanderService _service = IOverlanderService();

  IOverlanderRepository(this.db);

  // Carica da cache o da remoto
  Future<List<IOverlanderPlace>> getPlacesByCountry(String countryCode, {bool forceRefresh = false}) async {
    if (!forceRefresh) {
      final cached = await _getCachedPlaces(countryCode);
      if (cached.isNotEmpty) return cached;
    }

    // Scarica da remoto
    final remotePlaces = await _service.getCamperPlaces(countryCode);

    // Salva in cache
    await _cachePlaces(remotePlaces, countryCode);

    return remotePlaces;
  }

  Future<List<IOverlanderPlace>> _getCachedPlaces(String country) async {
    final query = db.select(db.iOverlanderPlaces)
      ..where((tbl) => tbl.country.equals(country))
      ..orderBy([(t) => OrderingTerm.desc(t.cachedAt)]);

    final results = await query.get();

    return results.map((row) => IOverlanderPlace(
      id: row.id,
      placeId: row.placeId,
      name: row.name,
      placeType: row.placeType,
      lat: row.latitude,
      lng: row.longitude,
      description: row.description,
      creator: row.creator,
      rating: row.rating,
    )).toList();
  }

  Future<void> _cachePlaces(List<IOverlanderPlace> places, String country) async {
    // Cancella vecchi dati per questo paese
    await (db.delete(db.iOverlanderPlaces)
          ..where((tbl) => tbl.country.equals(country)))
        .go();

    // Inserisci nuovi
    await db.batch((batch) {
      for (final place in places) {
        batch.insert(
          db.iOverlanderPlaces,
          IOverlanderPlacesCompanion.insert(
            placeId: place.placeId ?? DateTime.now().millisecondsSinceEpoch.toString(),
            name: place.name,
            placeType: place.placeType,
            latitude: place.lat,
            longitude: place.lng,
            description: Value(place.description),
            creator: Value(place.creator),
            rating: Value(place.rating),
            country: country,
          ),
        );
      }
    });
  }

  // Pulizia cache vecchia (> 30 giorni)
  Future<void> cleanOldCache() async {
    final cutoff = DateTime.now().subtract(const Duration(days: 30));
    await (db.delete(db.iOverlanderPlaces)
          ..where((tbl) => tbl.cachedAt.isSmallerThanValue(cutoff)))
        .go();
  }
}

    if (!serviceEnabled) return null;

    LocationPermission permission = await Geolocator.checkPermission();

2. Theme Mode ProviderCrea il file: lib/core/providers/theme_provider.dartdart

import 'package:flutter/material.dart';
import 'package:riverpod_annotation/riverpod_annotation.dart';
import 'package:shared_preferences/shared_preferences.dart';

part 'theme_provider.g.dart';

@riverpod
class ThemeModeNotifier extends _$ThemeModeNotifier {
  @override
  ThemeMode build() {
    _loadSavedTheme();
    return ThemeMode.system;
  }

  Future<void> _loadSavedTheme() async {
    final prefs = await SharedPreferences.getInstance();
    final savedTheme = prefs.getString('theme_mode');

    if (savedTheme == 'dark') {
      state = ThemeMode.dark;
    } else if (savedTheme == 'light') {
      state = ThemeMode.light;
    } else {
      state = ThemeMode.system;
    }
  }

  Future<void> setTheme(ThemeMode mode) async {
    state = mode;
    final prefs = await SharedPreferences.getInstance();
    String value = 'system';
    if (mode == ThemeMode.dark) value = 'dark';
    if (mode == ThemeMode.light) value = 'light';
    await prefs.setString('theme_mode', value);
  }
}

3. Subscription / RevenueCat ProviderCrea il file: lib/core/providers/subscription_provider.dartdart

import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:riverpod_annotation/riverpod_annotation.dart';
import 'package:purchases_flutter/purchases_flutter.dart';

part 'subscription_provider.g.dart';

@riverpod
class Subscription extends _$Subscription {
  @override
  Future<bool> build() async {
    try {
      final customerInfo = await Purchases.getCustomerInfo();
      return customerInfo.entitlements.active.isNotEmpty;
    } catch (e) {
      return false; // Se non Pro → false
    }
  }

  Future<void> refresh() async {
    state = const AsyncLoading();
    try {
      final customerInfo = await Purchases.getCustomerInfo();
      state = AsyncData(customerInfo.entitlements.active.isNotEmpty);
    } catch (e) {
      state = AsyncData(false);
    }
  }
}
